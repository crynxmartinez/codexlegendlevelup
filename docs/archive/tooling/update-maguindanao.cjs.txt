const GUILD = 'Mindanao Arch';
const DECK = 'Maguindanao';
const SUB = ['Maguindanao'];

function art(name) { return `assets/cards/web/maguindanao/${name}.webp`; }
function thumb(name) { return `assets/cards/thumb/maguindanao/${name}.webp`; }

function card(id, name, type, level, power, defense, state, abilities, text, flavor) {
  return {
    id, name, type, guild: GUILD, deck: DECK, level, power, defense,
    subtypes: type === 'Unit' ? SUB : [],
    state: state ?? null,
    abilities,
    text,
    flavor: flavor || '',
    artist: null,
    art: art(name),
    thumb: thumb(name),
    guildPrinted: GUILD,
  };
}
