// --- 2. Agta codex: add [red] Bayanihan as first paragraph ---
const agtaCodexIds = [
  'agta/hidden-path',
  'agta/listening-canopy',
  'agta/returning-river',
  'agta/shared-ember',
  'agta/thousand-knots',
];

for (const id of agtaCodexIds) {
  const card = cards.find(c => c.id === id);
  if (!card) { console.error('NOT FOUND:', id); continue; }
  let text = card.text;
  // Remove existing Bayanihan paragraph (with or without [red])
  text = text.replace(/(\[red\])?Bayanihan - [^\n]+\n?/g, '');
  // Split into paragraphs
  let paras = text.split('\n').filter(p => p.trim());
  // Insert [red] Bayanihan as first paragraph
  paras.unshift(BAYANIHAN);
  card.text = paras.join('\n');
  console.log('Updated Agta codex:', id);
}
