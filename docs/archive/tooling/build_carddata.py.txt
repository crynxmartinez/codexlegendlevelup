#!/usr/bin/env python3
"""
Generate src/data/cards.json (typed CardDef records) from cards.csv.

The JSON is what the game loads at runtime; cards.csv stays the human-editable
source of truth. Effect *logic* is hand-written per card in src/cards/ and keyed
by the same id - this file only carries stats and display text.
"""
from __future__ import annotations

import csv, json, re, sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
OUT = ROOT / "src" / "data" / "cards.json"

TYPE_MAP = {
    "Normal Unit": "NormalUnit",
    "Hero Unit": "HeroUnit",
    "Normal Action": "Action",
    "Sentry Action": "Sentry",
    "Force": "Force",
    "Codex": "Codex",
    "Token": "Token",
}

MARKER_TO_CATEGORY = {
    "blue": "Activated",
    "green": "Conditional",
    "pink": "Continuous",
    "yellow": "Battle",
    "red": "CodexPassive",
}

SCOPES = ["Ally", "Persona", "Hand", "Heart", "Grave", "Barrier", "Field"]
STATES = ["Grace", "Ravage", "Mourn", "Concert", "Ethereal", "Stampede",
          "Glory", "Overwhelm", "Demonic", "Ignite"]


def parse_abilities(effect: str) -> list[dict]:
    """Split effect text into ability records on the coloured markers."""
    if not effect.strip():
        return []
    parts = re.split(r"\[(blue|green|pink|yellow|red)\]", effect)
    out: list[dict] = []
    # parts[0] is any preamble (State definition line, reminder text)
    for i in range(1, len(parts), 2):
        marker, body = parts[i], parts[i + 1] if i + 1 < len(parts) else ""
        body = body.strip()
        tags = re.findall(r"\[([A-Za-z ]+)\]", body[:80])
        out.append({
            "category": MARKER_TO_CATEGORY[marker],
            "marker": marker,
            "scope": [t for t in tags if t in SCOPES],
            "requiresState": next((t for t in tags if t in STATES), None),
            "perTurn": 1 if re.search(r"1/\s*turn", body, re.I) else None,
            "text": body,
        })
    return out


def main() -> int:
    rows = list(csv.DictReader(open(ROOT / "cards.csv", encoding="utf-8-sig")))
    cards = []
    for r in rows:
        ctype = TYPE_MAP.get(r["card_type"])
        if not ctype:
            print(f"  ! unmapped card_type {r['card_type']!r} on {r['id']}")
            continue
        effect = r["effect"]
        cards.append({
            "id": r["id"],
            "name": r["name"],
            "type": ctype,
            "guild": r["guild"],
            "deck": r["deck"],
            "level": int(r["level"]) if r["level"] else 0,
            "power": int(r["power"]) if r["power"] else 0,
            "defense": int(r["defense"]) if r["defense"] else 0,
            "subtypes": [r["subtype"]] if r["subtype"] else [],
            "state": r["state"] or None,
            "abilities": parse_abilities(effect),
            "text": effect,
            "flavor": r["flavor"],
            "artist": r["artist"] or None,
            "art": r["webp"],
            "thumb": r["thumb"],
            "guildPrinted": r["guild_printed"],
        })

    # Engine Core is the one card that overrides the 3-copy rule.
    for c in cards:
        if c["id"] == "steam/engine-core":
            c["maxCopies"] = 8

    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(json.dumps(cards, indent=1, ensure_ascii=False), encoding="utf-8")

    angel = [c for c in cards if c["deck"] == "Angel"]
    print(f"wrote {OUT.relative_to(ROOT).as_posix()}  ({len(cards)} cards)")
    print(f"  Angel: {len(angel)} cards, "
          f"{sum(len(c['abilities']) for c in angel)} parsed abilities")
    no_abil = [c["name"] for c in angel if not c["abilities"] and c["type"] != "Token"]
    if no_abil:
        print(f"  Angel cards with no marker-tagged ability: {len(no_abil)}")
        for n in no_abil:
            print(f"    - {n}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
