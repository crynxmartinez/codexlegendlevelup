function agtaCard(id, name, type, level, power, defense, text, flavor) {
  return {
    id, name, type, guild: 'Gubat Banwa', deck: 'Agta', level, power, defense,
    subtypes: type === 'Unit' ? ['Luzon Arch'] : [],
    state: type === 'Unit' || type === 'Codex' ? 'Bayanihan' : null,
    abilities: [], text: text || '', flavor: flavor || '',
    artist: null,
    art: `assets/cards/web/Agta/${name}.webp`,
    thumb: `assets/cards/thumb/Agta/${name}.webp`,
    guildPrinted: 'Gubat Banwa',
  };
}

function magCard(id, name, type, level, power, defense, text, flavor) {
  return {
    id, name, type, guild: 'Mindanao Arch', deck: 'Maguindanao', level, power, defense,
    subtypes: type === 'Unit' ? ['Maguindanao'] : [],
    state: type === 'Unit' || type === 'Codex' ? 'Eclipse' : null,
    abilities: [], text: text || '', flavor: flavor || '',
    artist: null,
    art: `assets/cards/web/maguindanao/${name}.webp`,
    thumb: `assets/cards/thumb/maguindanao/${name}.webp`,
    guildPrinted: 'Mindanao Arch',
  };
}
