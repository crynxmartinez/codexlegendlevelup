#!/usr/bin/env python3
"""Append card rows (JSON on stdin) to the running cards.jsonl store."""
import json, sys, pathlib
store = pathlib.Path(sys.argv[1])
rows = json.load(sys.stdin)
with store.open("a", encoding="utf-8") as f:
    for r in rows:
        f.write(json.dumps(r, ensure_ascii=False) + "\n")
print(f"+{len(rows)} rows; total now {sum(1 for _ in store.open(encoding='utf-8'))}")
